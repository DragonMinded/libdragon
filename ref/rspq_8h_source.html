477inline rspq_write_t rspq_write_begin(uint32_t ovl_id, uint32_t cmd_id, int size) {
478 extern volatile uint32_t *rspq_cur_pointer, *rspq_cur_sentinel;
479 extern void rspq_next_buffer(void);
480
481 assertf(size <= RSPQ_MAX_COMMAND_SIZE, "The maximum command size is %d!", RSPQ_MAX_COMMAND_SIZE);
482
483 if (__builtin_expect(rspq_cur_pointer > rspq_cur_sentinel - size, 0))
484 rspq_next_buffer();
485
486 volatile uint32_t *cur = rspq_cur_pointer;
487 rspq_cur_pointer += size;
488
489 return (rspq_write_t){
490 .first_word = ovl_id + (cmd_id<<24),
491 .pointer = cur + 1,
492 .first = cur,
493 .is_first = 1
494 };
495}
